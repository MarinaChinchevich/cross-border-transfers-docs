// ============================================================
// Сервис авторизации 
// ============================================================

Table auth_services {
  service_id uuid [primary key, not null]
  service_name varchar(100) [not null, unique]
  service_secret_hash varchar(255) [not null]
  created_at timestamp [not null, default: now()]
}

// ============================================================
// Сервис переводов 
// ============================================================

Table transfers {
  transfer_id uuid [primary key, not null]
  user_id uuid [not null]
  amount decimal(15,2) [not null]
  currency_from char(3) [not null, default: 'RUB']
  currency_to char(3) [not null]
  exchange_rate decimal(10,6) [not null]
  commission decimal(10,2) [not null]
  total_amount decimal(15,2) [not null]
  status varchar(13) [not null]
  country_to char(2) [not null]
  beneficiary_id uuid [not null, ref: > beneficiaries.beneficiary_id ]
  payment_purpose varchar(255) [not null]
  hold_expires_at timestamp [not null]
  fx_rate_expires_at timestamp [not null]
  created_at timestamp [not null, default: now()]
}

Table transfer_statuses {
  transfer_status_id uuid [primary key, not null]
  transfer_id uuid [not null,  ref: > transfers.transfer_id]
  old_status varchar(13) [not null]
  new_status varchar(13) [not null]
  changed_at timestamp [not null, default: now()]
  reason text
}

Table idempotency_keys {
  idempotency_key_id uuid [primary key, not null]
  key varchar(255) [unique, not null]
  transfer_id uuid [not null,  ref: > transfers.transfer_id]
  expires_at timestamp [not null, default: now() + interval '24 hours']
  created_at timestamp [not null, default: now()]
}

Table user_limits {
  limit_id uuid [primary key, not null]
  user_id uuid [unique, not null]
  daily_limit decimal(15,2) [not null, default: 100000]
  monthly_limit decimal(15,2) [not null, default: 1000000]
  daily_spent decimal(15,2) [not null, default: 0]
  monthly_spent decimal(15,2) [not null, default: 0]
  updated_at timestamp [not null, default: now()]
}

Table beneficiaries {
  beneficiary_id uuid [primary key, not null]
  beneficiary_name varchar(255) [not null]
  user_id uuid [not null]
  account_number varchar(100) [not null]
  bic varchar(50) [not null]
  bank_name varchar(255) [not null]
  country char(2) [not null]
  created_at timestamp [not null, default: now()]
}

Table fx_rate_history {
  rate_id uuid [primary key, not null]
  currency_pair varchar(10) [not null]
  rate decimal(10,6) [not null]
  effective_rate decimal(10,6) [not null]
  created_at timestamp [not null, default: now()]
}


// ============================================================
// Сервис маршрутизации 
// ============================================================

Table routing_rules {
  rule_id uuid [primary key, not null]
  currency_pair varchar(10) [unique, not null]
  connector_type varchar(4) [not null]
}

Table routing_attempts {
  attempt_id uuid [primary key, not null]
  transfer_id uuid [not null]
  connector_type varchar(4) [not null]
  attempt_number integer [not null, default: 1]
  status varchar(50) [not null]
  error_message text
  created_at timestamp [not null, default: now()]
}

Table spfs_transactions {
  spfs_transaction_id uuid [primary key, not null]
  transfer_id uuid [not null]
  spfs_id varchar(255) [unique, not null]
  status varchar(50) [not null]
  payload_log jsonb [not null]
  error_code varchar(50)
  error_message text
  created_at timestamp [not null, default: now()]
  updated_at timestamp [not null, default: now()]
}

Table cips_transactions {
  cips_transaction_id uuid [primary key, not null]
  transfer_id uuid [not null]
  cips_id varchar(255) [unique, not null]
  status varchar(50) [not null]
  payload_log jsonb [not null]
  error_code varchar(50)
  error_message text
  created_at timestamp [not null, default: now()]
  updated_at timestamp [not null, default: now()]
}

// ============================================================
// Сервис возвратов
// ============================================================

Table refunds {
  refund_id uuid [primary key, not null]
  transfer_id uuid [not null]
  user_id uuid [not null]
  amount decimal(15,2) [not null]
  currency char(3) [not null]
  status varchar(50) [not null]
  reason text [not null]
  attempt_count integer [not null, default: 0]
  max_attempts integer [not null, default: 10]
  next_retry_at timestamp
  created_at timestamp [not null, default: now()]
  updated_at timestamp [not null, default: now()]
}

Table refund_attempts {
  attempt_id uuid [primary key, not null]
  refund_id uuid [not null, ref: > refunds.refund_id ]
  attempt_number integer [not null]
  status varchar(50) [not null]
  error_message text
  created_at timestamp [not null, default: now()]
}


// ============================================================
// Сервис уведомлений 
// ============================================================

Table notifications {
  notification_id uuid [primary key, not null]
  user_id uuid [not null]
  transfer_id uuid
  text varchar(50) [not null]
  status varchar(50) [not null, default: 'PENDING']
  error_message text
  sent_at timestamp
  created_at timestamp [not null, default: now()]
}
