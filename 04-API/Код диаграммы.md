@startuml

title: Создание перевода 

actor "Отправитель" as User
participant "Мобильное\nприложение" as App
participant "API Gateway" as GW
participant "МС Переводов" as Transfer
database "БД Переводов" as DB
participant "МС Курса\nвалют (SSE)" as FX
participant "Банк-\nотправитель" as Bank
participant "МС Комплаенс" as AML
queue "Kafka" as Kafka

autonumber

User -> App : Выбирает страну получателя и валюту перевода
activate App
App -> FX : SSE-подписка на курс 
activate FX

loop Обновление курса каждые 30 сек
    FX --> App : Обновление курса + пересчёт комиссии
end

deactivate FX

User -> App : Вводит сумму, реквизиты бенефициара, цель платежа\nНажимает "Подтвердить"
App -> GW : POST /transfers\n[Idempotency-Key]
activate GW
GW -> Transfer : Пересылает запрос
activate Transfer

Transfer -> DB : Проверка Idempotency-Key, лимита на переводы
activate DB
DB --> Transfer : ОК
deactivate DB

Transfer -> Bank : Запрос на холдирование 
activate Bank
Bank --> Transfer : Холд подтверждён
deactivate Bank

Transfer -> AML : Комплаенс-проверка 
activate AML
AML --> Transfer : risk_level = LOW
deactivate AML

Transfer -> DB : INSERT INTO transfers, transfer_statuses
activate DB
DB --> Transfer : OK
deactivate DB

Transfer -> Kafka : Публикация события transfer.created
activate Kafka
Kafka -->> Transfer : Подтверждение публикации
deactivate Kafka

Transfer --> GW : 201 Created {transfer_id, status, amount}
GW --> App : Данные перевода
App --> User : "Перевод создан"

alt Превышен лимит 
    Transfer -> DB : Проверка лимитов
    activate DB
    DB --> Transfer : Лимит исчерпан
    deactivate DB
    Transfer --> GW : 403 LIMIT_EXCEEDED
    GW --> App : Ошибка лимита
    App --> User : "Превышен лимит на переводы"
else Недостаточно средств 
    Transfer -> Bank : Запрос на холдирование
    activate Bank
    Bank --> Transfer : Ошибка INSUFFICIENT_FUNDS
    deactivate Bank
    Transfer --> GW : 402 INSUFFICIENT_FUNDS
    GW --> App : Ошибка
    App --> User : "Недостаточно средств"
else risk_level = HIGH 
    Transfer -> AML : Комплаенс-проверка
    activate AML
    AML --> Transfer : risk_level = HIGH
    deactivate AML
    Transfer -> Bank : Отмена холдирования
    activate Bank
    Bank --> Transfer : Холд отменён
    deactivate Bank
    Transfer -> DB : INSERT INTO transfers (status = BLOCKED)
    activate DB
    DB --> Transfer : OK
    deactivate DB
    Transfer --> GW : 201 Created {status: BLOCKED}
    GW --> App : Перевод заблокирован
    App --> User : "Перевод отклонён c указанием причины"
else risk_level = MEDIUM 
    Transfer -> AML : Комплаенс-проверка
    activate AML
    AML --> Transfer : risk_level = MEDIUM
    deactivate AML
    Transfer -> DB : INSERT INTO transfers (status = MANUAL_REVIEW)
    activate DB
    DB --> Transfer : OK
    deactivate DB
    Transfer --> GW : 201 Created {status: MANUAL_REVIEW}
    GW --> App : Перевод на проверке
    App --> User : "Перевод на проверке до 24 часов"
end
deactivate Transfer
deactivate GW
deactivate App

@enduml
